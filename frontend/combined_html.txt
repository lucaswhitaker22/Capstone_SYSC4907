                    <th onclick="sortTable(1)">Program</th>
                    <th onclick="sortTable(2)">Term</th>
                    <th onclick="sortTable(3)">Academic Year</th>
                    <th onclick="sortTable(4)">Block Size</th>
                    <th onclick="sortTable(5)">Rating</th>
                    <th onclick="sortTable(6)">Status</th>
                    <th>Actions</th>
                </tr>
            </thead>

            </tbody>

    <script>
        async function fetchBlocks() {
            try {
                const response = await fetch('http://127.0.0.1:5000/api/blocks');
                const blocks = await response.json();
                
                const tbody = document.getElementById('blocksBody');
                tbody.innerHTML = '';

                blocks.forEach(block => {
                    const row = document.createElement('tr');
                    row.innerHTML = `
                        <td>${block.block_id}</td>
                        <td>${block.program_id}</td>
                        <td>${block.term}</td>
                        <td>${block.academic_year}</td>
                        <td>${block.block_size} students</td>
                        <td>${block.schedule_rating ? block.schedule_rating + '/100' : 'Not rated'}</td>
                        <td><span class="status-badge status-${block.status}">${block.status}</span></td>
                        <td>
                            <button class="view-schedule" onclick="viewSchedule('${block.block_id}')">
                                View Schedule
                            </button>
                        </td>
                    `;
                    tbody.appendChild(row);
                });
            } catch (error) {
                console.error('Error fetching blocks:', error);
            }
        }

        function viewSchedule(blockId) {
            window.open(`schedule.html?blockId=${blockId}`, '_blank');
        }

        function sortTable(n) {
            const table = document.getElementById("blocksTable");
            let switching = true;
            let dir = "asc";
            let switchcount = 0;

            while (switching) {
                switching = false;
                const rows = table.rows;

                for (let i = 1; i < (rows.length - 1); i++) {
                    let shouldSwitch = false;
                    const x = rows[i].getElementsByTagName("TD")[n];
                    const y = rows[i + 1].getElementsByTagName("TD")[n];

                    if (dir === "asc") {
                        if (x.innerHTML.toLowerCase() > y.innerHTML.toLowerCase()) {
                            shouldSwitch = true;
                            break;
                        }
                    } else if (dir === "desc") {
                        if (x.innerHTML.toLowerCase() < y.innerHTML.toLowerCase()) {
                            shouldSwitch = true;
                            break;
                        }
                    }

                    if (shouldSwitch) {
                        rows[i].parentNode.insertBefore(rows[i + 1], rows[i]);
                        switching = true;
                        switchcount++;
                    }
                }

                if (switchcount === 0 && dir === "asc") {
                    dir = "desc";
                    switching = true;
                }
            }
        }

        document.addEventListener('DOMContentLoaded', fetchBlocks);
    </script>

    <script>
        function getDayName(day) {
            const days = ['Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday', 'Saturday','Sunday'];
            return days[day - 1];
        }

        function createTimeSlots() {
            const tbody = document.getElementById('calendarBody');
            const startHour = 8;
            const endHour = 18;
            
            for (let hour = startHour; hour < endHour; hour++) {
                const row = document.createElement('tr');
                row.innerHTML = `
                    <td class="time-header">${hour.toString().padStart(2, '0')}:00</td>
                    <td id="1-${hour}" class="time-slot"></td>
                    <td id="2-${hour}" class="time-slot"></td>
                    <td id="3-${hour}" class="time-slot"></td>
                    <td id="4-${hour}" class="time-slot"></td>
                    <td id="5-${hour}" class="time-slot"></td>
                `;
                tbody.appendChild(row);
            }
        }

        function placeOffering(offering) {
            const startHour = parseInt(offering.start_time.split(':')[0]);
            const startMin = parseInt(offering.start_time.split(':')[1]);
            const endHour = parseInt(offering.end_time.split(':')[0]);
            const endMin = parseInt(offering.end_time.split(':')[1]);
            
            const duration = (endHour - startHour) * 60 + (endMin - startMin);
            const topOffset = (startMin / 60) * 100;
            const height = (duration / 60) * 100;
            
            const cell = document.getElementById(`${offering.day_of_week}-${startHour}`);
            if (cell) {
                const courseBlock = document.createElement('div');
                courseBlock.className = `course-block ${offering.section_type}`;
                courseBlock.style.top = `${topOffset}%`;
                courseBlock.style.height = `${height}%`;
                courseBlock.innerHTML = `
                    ${offering.course_id}<br>
                    ${offering.section_type} ${offering.section_code}<br>
                    ${offering.start_time}-${offering.end_time}
                `;
                cell.appendChild(courseBlock);
            }
        }

        async function fetchBlockSchedule(blockId) {
            try {
                const [blockResponse, scheduleResponse] = await Promise.all([
                    fetch(`http://127.0.0.1:5000/api/blocks/${blockId}`),
                    fetch(`http://127.0.0.1:5000/api/blocks/${blockId}/schedule`)
                ]);

                const block = await blockResponse.json();
                const schedule = await scheduleResponse.json();
                
                document.getElementById('blockInfo').innerHTML = `
                    <h2>Block ${block.block_id}</h2>
                    <p>Program: ${block.program_id}</p>
                    <p>Term: ${block.term} ${block.academic_year}</p>
                    <p>Block Size: ${block.block_size}</p>
                    <p>Rating: ${block.schedule_rating || 'Not rated'}</p>
                `;

                createTimeSlots();
                
                // Populate both calendar and table
                schedule.forEach(offering => {
                    placeOffering(offering);
                    
                    const row = document.createElement('tr');
                    row.innerHTML = `
                        <td>${offering.course_id}</td>
                        <td>${offering.section_type}</td>
                        <td>${offering.section_code}</td>
                        <td>${getDayName(offering.day_of_week)}</td>
                        <td>${offering.start_time}-${offering.end_time}</td>
                    `;
                    document.getElementById('scheduleBody').appendChild(row);
                });
            } catch (error) {
                console.error('Error:', error);
            }
        }

        document.addEventListener('DOMContentLoaded', () => {
            const urlParams = new URLSearchParams(window.location.search);
            const blockId = urlParams.get('blockId') || 'F_ENG_1';
            fetchBlockSchedule(blockId);
        });
    </script>
